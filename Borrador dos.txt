CREATE OR REPLACE TRIGGER TRG_VIGILAR_INGLES AFTER UPDATE OF CODIGO_INGLES ON TABLA_PRUEBA 
FOR EACH ROW
  BEGIN
    --: OLD TRAE EL DATO COMO ESTABA ANTES
    INSERT INTO LOG_AUDITORIA_ESTUDIANTE
    (FECHA_CAMBIO, USUARIO_QUE_CAMBIO, CEDULA_AFECTADA, DATO_VIEJO, DATO_NUEVO)
    VALUES
    (SYSDATE, USER, :OLD.NUM_IDENTIFICACION, :OLD.CODIGO_INGLES, :NEW.CODIGO_INGLES);
  
    END;
    



UPDATE TABLA_PRUEBA
       SET CODIGO_INGLEWS = 'INGLES_NIVEL_4'
       WHERE NUM_IDENTIFICACION = '52278452';
COMMIT;


 CREATE TABLE LOG_EJECUCIONES_MASIVAS(
 FECHA_EJECUCION DATE,
 USUARIO_RED VARCHAR2(50),
 GRUPO_ASIGNADO VARCHAR2(50),
 CANTIDAD_REGISTROS VARCHAR2(50),
 CANTIDAD_REGISTROS NUMBER,
 ESTADO_PROCESO VARCHAR2(100)
 );




CREATE OR REPLACE PROCEDURE ACTUALIZAR_GRUPO_INGLES (
    p_nuevo_grupo IN VARCHAR2
) AS
   v_filas_afectadas NUMBER;
BEGIN
   -- 1. Hacemos la actualización masiva en la tabla oficial
   UPDATE TABLA_PRUEBA
   SET CODIGO_INGLES = p_nuevo_grupo
   WHERE NUM_IDENTIFICACION IN (SELECT NUM_IDENTIFICACION FROM TEMP_INGLES);
   
   -- 2. Oracle nos dice cuántos estudiantes se acaban de actualizar
   v_filas_afectadas := SQL%ROWCOUNT;
   
   -- 3. Guardamos tu "Log de Auditoría de Proceso"
   INSERT INTO LOG_EJECUCIONES_MASIVAS 
   (FECHA_EJECUCION, USUARIO_RED, GRUPO_ASIGNADO, CANTIDAD_REGISTROS, ESTADO_PROCESO)
   VALUES 
   (SYSDATE, USER, p_nuevo_grupo, v_filas_afectadas, 'EXITOSO');
   
   -- 4. Confirmamos todo
   COMMIT;
   DBMS_OUTPUT.PUT_LINE('Proceso terminado. Se actualizaron ' || v_filas_afectadas || ' estudiantes.');

EXCEPTION
   -- 5. Si algo explota, también lo anotamos en el log para saber por qué falló
   WHEN OTHERS THEN
      ROLLBACK;
      INSERT INTO LOG_EJECUCIONES_MASIVAS 
      VALUES (SYSDATE, USER, p_nuevo_grupo, 0, 'FALLO: ' || SUBSTR(SQLERRM, 1, 80));
      COMMIT;
      DBMS_OUTPUT.PUT_LINE('Error en el proceso masivo.');
END ACTUALIZAR_GRUPO_INGLES;
